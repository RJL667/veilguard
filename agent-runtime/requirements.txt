# ──────────────────────────────────────────────────────────────────────────
# Veilguard agent-runtime — Phase 0 dependencies
# ──────────────────────────────────────────────────────────────────────────
# The Claude Agent SDK is the agent loop runtime.  This service wraps it
# with our middleware (TCMM render, audit, approval-gate hook, tenant
# context) and exposes a streaming SSE endpoint LibreChat (via pii-proxy)
# can call.
#
# Pinning notes:
#   - lancedb pinned to the same version pii-proxy + TCMM use.  Mixed
#     versions writing to the same on-disk store would risk version skew.
#   - pyarrow >= 15 matches TCMM (pii-proxy uses the same floor).
#   - fastapi / uvicorn match pii-proxy so we're not running two webserver
#     stacks with different perf profiles.

fastapi==0.115.6
uvicorn[standard]==0.34.0
httpx==0.28.1

# 2026-05-25 (Phase R cleanup): claude-agent-sdk is no longer required.

# pii-proxy's ChatAgent), using TCMM's OAuth-bearer adapter instead of
# the SDK's own session manager.  Leaving the pin commented out so
# anyone resurrecting the legacy SDK path knows the version that was
# in place last.
# claude-agent-sdk>=0.0.20

# Decision-ledger storage (agent_tasks, task_proposals, org_memory, etc.)
# Same Lance dir as TCMM + pii_audit — bump in lockstep with both.
lancedb==0.30.2
pyarrow>=15.0.0

# 2026-05-25: agent-runtime now mounts and imports the shared `agent/`
# package (the 5-step pipeline) from the repo root via PYTHONPATH=/.

# presidio + spacy at the EXACT versions pii-proxy uses; mismatched

presidio-analyzer==2.2.356
presidio-anonymizer==2.2.356

# Anthropic SDK — TCMM's loaded adapter calls anthropic.Anthropic(
